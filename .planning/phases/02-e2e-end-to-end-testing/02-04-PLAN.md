The file must follow the SDK compilation verification pattern from the research (Pattern 5). Key insight: The generated TypeScript SDK files import types from `@synapse-router/core` which may not be resolvable from a standalone temp directory. The approach is to generate a minimal tsconfig with `paths` aliases and `skipLibCheck: true`.

```typescript
import { describe, it } from "node:test";
import assert from "node:assert/strict";
import { generateSdk } from "../../packages/sdk-generator/src/generator.js";
import { execSync } from "node:child_process";
import { writeFileSync } from "node:fs";
import { join } from "node:path";
import { ROOT, PACKS_DIR, withTempDir } from "./helpers.js";
```

**Test structure:**

```typescript
const OPENAI_PACK = join(PACKS_DIR, "openai");

describe("SDK compilation verification E2E", () => {
  describe("TypeScript SDK compilation", () => {
    it("generated TypeScript SDK compiles with tsc --noEmit", async () => {
      const { dir, cleanup } = withTempDir();
      try {
        // Generate SDK to temp directory
        const result = await generateSdk({
          packDir: OPENAI_PACK,
          lang: "typescript",
          outputDir: dir,
        });
        assert.ok(result.files.length > 0,
          "should have generated files");

        // Create a minimal tsconfig that can resolve @synapse-router/core
        // using paths aliases pointing to the monorepo source
        writeFileSync(join(dir, "tsconfig.json"), JSON.stringify({
          compilerOptions: {
            target: "ES2022",
            module: "NodeNext",
            moduleResolution: "NodeNext",
            strict: true,
            noEmit: true,
            skipLibCheck: true,
            paths: {
              "@synapse-router/core":
                [join(ROOT, "packages/core/src/index.ts")]
            },
            baseUrl: dir,
          },
          include: ["*.ts"],
        }));

        // Run tsc --noEmit from the temp directory
        execSync("tsc --noEmit", {
          cwd: dir,
          stdio: "pipe",
          env: {
            ...process.env,
            PATH: join(ROOT, "node_modules/.bin") + ":" +
                  (process.env.PATH || ""),
          },
        });
      } finally {
        cleanup();
      }
    });
  });

  describe("Python SDK syntax verification", () => {
    it("generated Python SDK parses without syntax errors", async () => {
      const { dir, cleanup } = withTempDir();
      try {
        const result = await generateSdk({
          packDir: OPENAI_PACK,
          lang: "python",
          outputDir: dir,
        });
        assert.ok(result.files.length > 0,
          "should have generated files");

        // Check each .py file for valid syntax
        for (const file of result.files) {
          if (file.path.endsWith(".py")) {
            const filePath = join(dir, file.path);
            execSync(
              `python3 -c "compile(open('${filePath}').read(), ` +
              `'${file.path}', 'exec')"`,
              { stdio: "pipe" }
            );
          }
        }
      } finally {
        cleanup();
      }
    });
  });

  describe("SDK output structure", () => {
    it("TypeScript SDK generates expected file structure", async () => {
      const { dir, cleanup } = withTempDir();
      try {
        const result = await generateSdk({
          packDir: OPENAI_PACK,
          lang: "typescript",
          outputDir: dir,
        });
        const paths = result.files.map((f) => f.path);
        assert.ok(paths.includes("index.ts"),
          "should include index.ts");
        assert.ok(paths.includes("client.ts"),
          "should include client.ts");
        assert.ok(paths.includes("types.ts"),
          "should include types.ts");
        assert.ok(paths.includes("package.json"),
          "should include package.json");
        assert.ok(paths.includes("tsconfig.json"),
          "should include tsconfig.json");
        assert.ok(paths.includes("README.md"),
          "should include README.md");
      } finally {
        cleanup();
      }
    });

    it("Python SDK generates expected file structure", async () => {
      const { dir, cleanup } = withTempDir();
      try {
        const result = await generateSdk({
          packDir: OPENAI_PACK,
          lang: "python",
          outputDir: dir,
        });
        const paths = result.files.map((f) => f.path);
        assert.ok(paths.includes("__init__.py"),
          "should include __init__.py");
        assert.ok(paths.includes("client.py"),
          "should include client.py");
        assert.ok(paths.includes("pyproject.toml"),
          "should include pyproject.toml");
        assert.ok(paths.includes("README.md"),
          "should include README.md");
      } finally {
        cleanup();
      }
    });
  });
});
```

**Important details about tsc resolution (addressing research Assumption A4):**

The generated TypeScript SDK files import from `@synapse-router/core`. The tsconfig must have `paths` configured to resolve this import:
```json
"paths": {
  "@synapse-router/core": ["<absolute-path-to>/packages/core/src/index.ts"]
}
```
This approach works because:
1. `skipLibCheck: true` avoids type-checking issues in .d.ts files
2. The `paths` mapping lets tsc find the @synapse-router/core types in the monorepo
3. `baseUrl` is set to the output directory so relative generated imports work
4. The `env.PATH` is overridden to include the project root's `node_modules/.bin` so `tsc` can be found

**Python verification:** Use `python3 -c "compile(...)"` to check syntax only (no execution). This catches Python syntax errors without requiring dependencies.

**Cleanup:** Both tests use `withTempDir()` from helpers and have `try/finally` blocks.

**Timeout:** Set timeout of 30000ms for the tsc compilation test since it involves file generation and compilation.
