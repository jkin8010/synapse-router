**withMcpServer helper:**

```typescript
import { spawn } from "node:child_process";
import { join } from "node:path";
import { ROOT } from "./helpers.js";

async function withMcpServer(
  fn: (send: (msg: object) => Promise<object>) => Promise<void>
): Promise<void> {
  const bin = join(ROOT, "apps", "mcp-server", "dist", "index.js");
  const proc = spawn("node", [bin], { stdio: ["pipe", "pipe", "pipe"] });
  const pending = new Map<string | number, (data: unknown) => void>();

  let buffer = "";
  const stdout = proc.stdout!;
  stdout.on("data", (chunk: Buffer) => {
    buffer += chunk.toString();
    const lines = buffer.split("\n");
    buffer = lines.pop() ?? "";
    for (const line of lines) {
      if (!line.trim()) continue;
      try {
        const parsed = JSON.parse(line);
        const resolve = pending.get(parsed.id);
        if (resolve) {
          pending.delete(parsed.id);
          resolve(parsed);
        }
      } catch {
        // skip partial lines
      }
    }
  });

  const send = (msg: object): Promise<object> => {
    return new Promise((resolve) => {
      const id = (msg as any).id;
      pending.set(id, resolve);
      proc.stdin!.write(JSON.stringify(msg) + "\n");
    });
  };

  try {
    // Give the server a moment to start
    await new Promise((r) => setTimeout(r, 200));
    await fn(send);
  } finally {
    proc.kill();
    // Force kill if still alive after 500ms
    setTimeout(() => {
      try { proc.kill("SIGKILL"); } catch {}
    }, 500);
  }
}
```

**Test structure:**

```typescript
import { describe, it } from "node:test";
import assert from "node:assert/strict";

describe("MCP Server negotiate tool E2E", () => {
  it("initialize -> tools/list -> tools/call negotiate completes", async () => {
    await withMcpServer(async (send) => {
      // Step 1: Initialize
      const initResult = await send({
        jsonrpc: "2.0", id: 1, method: "initialize", params: {},
      });
      const init = initResult as any;
      assert.equal(init.result?.protocolVersion, "2024-11-05",
        `Expected protocolVersion, got: ${JSON.stringify(init.result)}`);
      assert.ok(init.result?.capabilities?.tools,
        "should have tools capability");

      // Step 2: tools/list
      const listResult = await send({
        jsonrpc: "2.0", id: 2, method: "tools/list", params: {},
      });
      const list = listResult as any;
      assert.ok(Array.isArray(list.result?.tools),
        "should have tools array");
      const toolNames = list.result.tools.map((t: any) => t.name);
      assert.ok(toolNames.includes("negotiate"),
        "negotiate tool should be listed");

      // Step 3: tools/call negotiate
      const result = await send({
        jsonrpc: "2.0", id: 3, method: "tools/call",
        params: {
          name: "negotiate",
          arguments: {
            agent_schema: {
              task: "generate chat response",
              goal: "reply to user query",
            },
            provider_id: "openai",
          },
        },
      });
      const neg = result as any;
      // The negotiate tool may fail if packs are not found, but should respond
      assert.ok(neg.result?.content?.[0]?.text !== undefined,
        "should have content text in negotiate response");
    });
  });

  it("tools/call with missing provider_id returns error", async () => {
    await withMcpServer(async (send) => {
      // Initialize first
      await send({
        jsonrpc: "2.0", id: 1, method: "initialize", params: {},
      });

      // Call negotiate without provider_id
      const result = await send({
        jsonrpc: "2.0", id: 2, method: "tools/call",
        params: {
          name: "negotiate",
          arguments: {
            agent_schema: { task: "test" },
          },
        },
      });
      const neg = result as any;
      // Per tools.ts: negotiate inputSchema has required: ["provider_id"]
      // Missing provider_id should trigger error in MCP handler
      assert.ok(
        neg.error !== undefined ||
        (neg.result?.isError === true),
        "Missing provider_id should result in error"
      );
    });
  });
});
```

**Key behavioral note:** The `negotiateAndVerify` function (verifier.ts:202-294) will:
1. Call `negotiateSchema` with mock LLM -> returns schema with 1 field ("input": "string")

6. Return final result with schema, schema_id, sdk_code, verification_result

The mock schema has one required field "input" of type "string". `runMockTest` creates a request body with `{ input: "test_input" }`. The MockServer validates this against the schema, so it should return 200.
