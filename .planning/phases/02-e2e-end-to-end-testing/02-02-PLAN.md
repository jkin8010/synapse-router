describe("CLI connect E2E", () => {
  it("synapse connect openai chat completes full flow with MockServer", async () => {
    // 1. Start MockServer on random port
    const { server, port } = await startMockServer();
    try {
      // 2. Construct args dynamically with the mock port
      const mockInput = JSON.stringify({
        messages: [{ role: "user", content: "hello" }],
        model: "gpt-4"
      });
      const args = [
        "connect",
        "openai",
        "chat",
        "--input", mockInput,
        "--base-url", `http://localhost:${port}`,
        "--api-key", "sk-placeholder",
        "--format", "json"
      ];
      // 3. Run CLI
      const result = runCli(args);
      // 4. Assert CLI exits cleanly
      assert.ok(result.stderr.length === 0 || result.stderr.includes("not yet"),
        `stderr should be empty or contain expected message, got: ${result.stderr}`);
      // stdout should contain JSON output (either ConnectResult or debug info)
      const parsed = JSON.parse(result.stdout);
      assert.ok(typeof parsed === "object", "stdout should be parseable JSON");
    } finally {
      await server.stop();
    }
  });

  it("synapse connect with --help prints help text", () => {
    const result = runCli(["connect", "--help"]);
    assert.ok(result.stdout.includes("synapse connect"),
      `help text should mention connect, got: ${result.stdout}`);
    assert.ok(result.stdout.includes("Usage:"),
      `help text should contain Usage:, got: ${result.stdout}`);
    assert.strictEqual(result.code, 0);
  });

  it("synapse connect without args starts interactive mode (no crash)", () => {
    const result = runCli(["connect"]);
    // Interactive mode reads from stdin -- without stdin it will hang or error.
    // We just check it doesn't crash.
    assert.ok(result.code === 0 || result.code === 1,
      `exit code should be 0 or 1, got ${result.code}`);
  });

```typescript
async function withServer(
  options: ServerOptions,
  fn: (baseUrl: string, server: http.Server) => Promise<void>
): Promise<void> {
  const server = createServer(options) as http.Server;
  await new Promise<void>((resolve) => server.listen(0, resolve));
  const addr = server.address();
  const baseUrl = `http://localhost:${(addr as any).port}`;
  if ((server as any)._readyPromise) {
    await (server as any)._readyPromise;
  }
  try {
    await fn(baseUrl, server);
  } finally {
    await new Promise<void>((resolve) => server.close(() => resolve()));
  }
}
```

```typescript
describe("API /negotiate E2E", () => {
  let vault: ReturnType<typeof createTempVault>;

  before(() => {
    vault = createTempVault();
  });

  after(() => {
    vault.cleanup();
  });

  it("POST /negotiate with agent_schema returns 200 with negotiation results", async () => {
    await withServer(getTestOptions(vault.vaultDir), async (baseUrl) => {
      const res = await httpRequest(`${baseUrl}/negotiate`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: { agent_schema: CHAT_AGENT_SCHEMA },
      });
      assert.equal(res.status, 200,
        `Expected 200, got ${res.status}: ${JSON.stringify(res.body)}`);
      const body = res.body as Record<string, unknown>;
      assert.ok(body.negotiation_id, "should have negotiation_id");
      assert.ok(Array.isArray(body.accepted), "should have accepted array");
      assert.ok(Array.isArray(body.rejected), "should have rejected array");
    });
  });

  it("POST /negotiate with provider_id filters to that provider", async () => {
    await withServer(getTestOptions(vault.vaultDir), async (baseUrl) => {
      const res = await httpRequest(`${baseUrl}/negotiate`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: {
          agent_schema: CHAT_AGENT_SCHEMA,
          provider_id: "openai",
        },
      });
      assert.equal(res.status, 200);
      const body = res.body as { accepted?: Array<{ provider_id?: string }> };
      assert.ok(body.accepted && body.accepted.length > 0,
        "should have at least one accepted provider");
      if (body.accepted && body.accepted.length > 0) {
        assert.equal(body.accepted[0].provider_id, "openai",
          "first accepted should be openai");
      }
    });
  });

  it("POST /negotiate without agent_schema returns 400", async () => {
    await withServer(getTestOptions(vault.vaultDir), async (baseUrl) => {
      const res = await httpRequest(`${baseUrl}/negotiate`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: {},
      });
      assert.equal(res.status, 400);
      assert.equal((res.body as any).error?.code, "INVALID_REQUEST");
    });
  });
});
```
