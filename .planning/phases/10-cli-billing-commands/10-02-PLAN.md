## Interface Contracts

### billing-config.json schema (defined by Plan 10-01):
```json
{
  "version": "1",
  "providers": {
    "<providerId>": {
      "pricing": { "<featureId>": { PricingRule } },
      "quotas": [ { Quota } ],
      "onError": "warn" | "reject"
    }
  }
}
```

### FileQuotaStorage file format (~/.synapse/quotas/<providerId>.json):
```json
{
  "<quotaId>": {
    "id": "<providerId>-cost-quota",
    "type": "cost",
    "limit": 10,
    "used": 3.5,
    "period": "month",
    "threshold": 0.8,
    "resetAt": "2026-06-01T00:00:00.000Z"
  }
}
```
Stored as `Record<string, Quota>` where key matches quota.id value.

### From packages/billing/src/types.ts:
```typescript
type QuotaPeriod = 'minute' | 'hour' | 'day' | 'month' | 'oneTime';
type PricingStrategy = 'perToken' | 'perUnit' | 'tiered' | 'flat';
type BillingOnError = 'warn' | 'reject' | 'downgrade' | 'notify';
interface Quota { id: string; type: string; limit: number; used: number; period: QuotaPeriod; threshold?: number; resetAt?: string; }
interface PricingRule { strategy: PricingStrategy; unit?: string; inputPrice?: number; outputPrice?: number; unitPrice?: number; costPerCall?: number; }
```

```typescript
import { readFile, writeFile, mkdir } from "node:fs/promises";
import { homedir } from "node:os";
import { join } from "node:path";

const SYNAPSE_DIR = join(homedir(), ".synapse");
const BILLING_CONFIG_PATH = join(SYNAPSE_DIR, "billing-config.json");

interface BillingConfigFile {
  version: string;
  providers: Record<string, {
    pricing: Record<string, object>;
    quotas: object[];
    onError: string;
  }>;
}

async function loadBillingConfig(): Promise<BillingConfigFile> {
  try {
    const content = await readFile(BILLING_CONFIG_PATH, "utf8");
    return JSON.parse(content);
  } catch {
    return { version: "1", providers: {} };
  }
}

async function saveBillingConfig(config: BillingConfigFile): Promise<void> {
  await mkdir(SYNAPSE_DIR, { recursive: true });
  await writeFile(BILLING_CONFIG_PATH, JSON.stringify(config, null, 2) + "\n", "utf8");
}
```
